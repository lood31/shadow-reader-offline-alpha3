cmake_minimum_required(VERSION 3.22)
project(ShadowNativeDependencies VERSION 1.52.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(SHADOW_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
get_filename_component(REPO_ROOT "${SHADOW_ROOT}/.." ABSOLUTE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
add_subdirectory("${REPO_ROOT}/third_party/whisper.cpp-1.9.4" whisper)
set(ESPEAK_ROOT "${REPO_ROOT}/third_party/espeak-ng-1.52.0")
set(USE_ASYNC OFF)
set(USE_KLATT OFF)
set(USE_LIBPCAUDIO OFF)
set(USE_LIBSONIC OFF)
set(USE_MBROLA OFF)
set(USE_SPEECHPLAYER OFF)
set(HAVE_MKSTEMP ON)
add_library(espeak-include INTERFACE)
target_include_directories(espeak-include INTERFACE "${ESPEAK_ROOT}/src/include")
add_subdirectory("${ESPEAK_ROOT}/src/ucd-tools" ucd)
add_subdirectory("${ESPEAK_ROOT}/src/libespeak-ng" espeak)
target_include_directories(espeak-ng BEFORE PRIVATE "${ESPEAK_ROOT}/src/include/compat")
target_compile_options(espeak-ng PRIVATE "-include" "${CMAKE_CURRENT_LIST_DIR}/apple_endian.h")
